// Пересчет сумм перед записью документа
// Гарантирует целостность данных: сумма строки и общая сумма документа 
// всегда рассчитываются автоматически, исключая ошибки ручного ввода или программной загрузки.
Процедура ПередЗаписью(Отказ, ПараметрыЗаписи, РежимПроведения)
    
    СуммаДокумента = 0;
    Для Каждого Строка Из Продукция Цикл
        Строка.Сумма = Строка.Количество * Строка.Себестоимость;
        СуммаДокумента = СуммаДокумента + Строка.Сумма;
    КонецЦикла;
    
КонецПроцедуры

// Обработка проведения: формирование движений по регистру накопления
// КЛЮЧЕВАЯ ОСОБЕННОСТЬ: использование запроса с СГРУППИРОВАТЬ ПО.
// Это предотвращает создание дублирующихся записей в регистре, если в табличной 
// части документа одна и та же номенклатура указана в нескольких разных строках.
Процедура ОбработкаПроведения(Отказ, Режим)
    
    // 1. Сворачиваем одинаковую номенклатуру через запрос
    Запрос = Новый Запрос;
    Запрос.Текст = 
    "ВЫБРАТЬ
    |    Продукция.Номенклатура КАК Номенклатура,
    |    СУММА(Продукция.Количество) КАК Количество
    |ИЗ
    |    Документ.ВыпускПродукции.Продукция КАК Продукция
    |ГДЕ
    |    Продукция.Ссылка = &Ссылка
    |СГРУППИРОВАТЬ ПО
    |    Продукция.Номенклатура";
    
    Запрос.УстановитьПараметр("Ссылка", Ссылка);
    Выборка = Запрос.Выполнить().Выбрать();
    
    // 2. Записываем уже свернутые (агрегированные) данные в регистр
    Движения.ОстаткиНоменклатуры.Записывать = Истина;
    
    Пока Выборка.Следующий() Цикл
        Движение = Движения.ОстаткиНоменклатуры.Добавить();
        Движение.ВидДвижения  = ВидДвиженияНакопления.Приход;
        Движение.Период       = Дата;
        Движение.Номенклатура = Выборка.Номенклатура;
        Движение.Количество   = Выборка.Количество;
    КонецЦикла;
    
КонецПроцедуры
