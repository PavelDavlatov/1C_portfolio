// 1. Автоматический пересчет сумм перед сохранением
// Гарантирует математическую целостность документа, исключая ошибки 
// ручного ввода или некорректной программной загрузки данных.
Процедура ПередЗаписью(Отказ, ПараметрыЗаписи, РежимПроведения)
    
    СуммаДокумента = 0;
    Для Каждого Строка Из Товары Цикл
        Строка.Сумма = Строка.Количество * Строка.Цена;
        СуммаДокумента = СуммаДокумента + Строка.Сумма;
    КонецЦикла;
    
КонецПроцедуры

// 2. Строгая валидация данных с указанием пользователю точного места ошибки
// Используется индексный цикл для точного определения номера проблемной строки.
// Свойство "Поле" объекта СообщениеПользователю подсвечивает ячейку в интерфейсе, 
// что значительно улучшает пользовательский опыт и ускоряет исправление ошибок.
Процедура ОбработкаПроверкиЗаполнения(Отказ, ПроверяемыеРеквизиты)
    
    Для Индекс = 0 По Товары.Количество() - 1 Цикл
        Строка = Товары[Индекс];
        Если НЕ ЗначениеЗаполнено(Строка.Сумма) Тогда
            Отказ = Истина; // Запрещаем запись документа
            
            Сообщение = Новый СообщениеПользователю;
            Сообщение.Текст = "В строке " + (Индекс + 1) + " не заполнена сумма!";
            // Ключевой момент: привязка сообщения к конкретной ячейке табличной части
            Сообщение.Поле  = "Товары[" + Индекс + "].Сумма";
            Сообщение.Сообщить();
        КонецЕсли;
    КонецЦикла;
    
КонецПроцедуры
