Процедура ЗаполнитьБазуНаСервере()
    
    // Создаём генератор случайных чисел
    Генератор = Новый ГенераторСлучайныхЧисел();
    
    // === 1. Создаём 100 номенклатур ===
    Для і = 1 По 100 Цикл
        НоваяНоменклатура = Справочники.Номенклатура.СоздатьЭлемент();
        НоваяНоменклатура.Наименование = "Товар" + і;
        НоваяНоменклатура.Цена = 100 + і;
        НоваяНоменклатура.Записать();
    КонецЦикла;
    Сообщить("Создано 100 номенклатур");
    
    // === 2. Проверяем наличие складов и контрагентов ===
    Склады = ПолучитьСписокСкладов();
    Если Склады.Количество() = 0 Тогда
        Сообщить("Сначала создайте хотя бы один склад!");
        Возврат;
    КонецЕсли;
    
    Контрагенты = ПолучитьСписокКонтрагентов();
    Если Контрагенты.Количество() = 0 Тогда
        Сообщить("Сначала создайте хотя бы одного контрагента!");
        Возврат;
    КонецЕсли;
    
    Номенклатуры = ПолучитьСписокНоменклатуры();
    
    // === 3. Создаём 200 приходных накладных ===
    Для і = 1 По 200 Цикл
        СоздатьПриходнуюНакладную(Номенклатуры, Склады, Контрагенты, Генератор);
    КонецЦикла;
    Сообщить("Создано 200 приходных накладных");
    
    // === 4. Создаём 300 расходных накладных ===
    Для і = 1 По 300 Цикл
        СоздатьРасходнуюНакладную(Номенклатуры, Склады, Контрагенты, Генератор);
    КонецЦикла;
    Сообщить("Создано 300 расходных накладных");
    
    Сообщить("Готово! База заполнена.");
    
КонецПроцедуры

    СлучайныйДень = Генератор.СлучайноеЧисло(0, 364);
