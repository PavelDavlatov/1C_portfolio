// Создание и проведение приходной накладной
// Директива &НаСервере обязательна, так как происходит прямая запись в базу данных
&НаСервере
Процедура СоздатьПриходнуюНакладную(Номенклатуры, Склады, Контрагенты, Генератор)
    
    // 1. Создаем новый объект документа в памяти
    Док = Документы.ПриходнаяНакладная.СоздатьДокумент();
    
    // 2. Заполняем шапку случайными, но валидными ссылками из переданных массивов
    Док.Дата       = ПолучитьСлучайнуюДату(Генератор);
    Док.Склад      = Склады[Генератор.СлучайноеЧисло(0, Склады.Количество() - 1)];
    Док.Контрагент = Контрагенты[Генератор.СлучайноеЧисло(0, Контрагенты.Количество() - 1)];
    
    // 3. Формируем табличную часть (случайное количество строк от 1 до 5)
    КоличествоСтрок = Генератор.СлучайноеЧисло(1, 5);
    Для к = 1 По КоличествоСтрок Цикл
        НоваяСтрока = Док.Товары.Добавить();
        НоваяСтрока.Номенклатура = Номенклатуры[Генератор.СлучайноеЧисло(0, Номенклатуры.Количество() - 1)];
        НоваяСтрока.Количество   = Генератор.СлучайноеЧисло(1, 100);
        НоваяСтрока.Цена         = 100 + Генератор.СлучайноеЧисло(0, 500);
        // Автоматический расчет суммы для обеспечения целостности данных
        НоваяСтрока.Сумма        = НоваяСтрока.Количество * НоваяСтрока.Цена;
    КонецЦикла;
    
    // 4. Сразу проводим документ, чтобы корректно сформировались движения по регистрам
    Док.Записать(РежимЗаписиДокумента.Проведение);
    
КонецПроцедуры

// Создание и проведение расходной накладной
&НаСервере
Процедура СоздатьРасходнуюНакладную(Номенклатуры, Склады, Контрагенты, Генератор)
    
    Док = Документы.РасходнаяНакладная.СоздатьДокумент();
    Док.Дата       = ПолучитьСлучайнуюДату(Генератор);
    Док.Склад      = Склады[Генератор.СлучайноеЧисло(0, Склады.Количество() - 1)];
    Док.Контрагент = Контрагенты[Генератор.СлучайноеЧисло(0, Контрагенты.Количество() - 1)];
    
    КоличествоСтрок = Генератор.СлучайноеЧисло(1, 5);
    Для к = 1 По КоличествоСтрок Цикл
        НоваяСтрока = Док.Товары.Добавить();
        НоваяСтрока.Номенклатура = Номенклатуры[Генератор.СлучайноеЧисло(0, Номенклатуры.Количество() - 1)];
        НоваяСтрока.Количество   = Генератор.СлучайноеЧисло(1, 5); // Меньший расход для контроля остатков
        НоваяСтрока.Цена         = 150 + Генератор.СлучайноеЧисло(0, 500);
        НоваяСтрока.Сумма        = НоваяСтрока.Количество * НоваяСтрока.Цена;
    КонецЦикла;
    
    Док.Записать(РежимЗаписиДокумента.Проведение);
    
КонецПроцедуры
