Процедура СоздатьПриходнуюНакладную(Номенклатуры, Склады, Контрагенты, Генератор)
    
    Док = Документы.ПриходнаяНакладная.СоздатьДокумент();
    Док.Дата       = ПолучитьСлучайнуюДату(Генератор);
    Док.Склад      = Склады[Генератор.СлучайноеЧисло(0, Склады.Количество() - 1)];
    Док.Контрагент = Контрагенты[Генератор.СлучайноеЧисло(0, Контрагенты.Количество() - 1)];
    
    КоличествоСтрок = Генератор.СлучайноеЧисло(1, 5);
    Для к = 1 По КоличествоСтрок Цикл
        НоваяСтрока = Док.Товары.Добавить();
        НоваяСтрока.Номенклатура = Номенклатуры[Генератор.СлучайноеЧисло(0, Номенклатуры.Количество() - 1)];
        НоваяСтрока.Количество   = Генератор.СлучайноеЧисло(1, 100);
        НоваяСтрока.Цена         = 100 + Генератор.СлучайноеЧисло(0, 500);
        НоваяСтрока.Сумма        = НоваяСтрока.Количество * НоваяСтрока.Цена;
    КонецЦикла;
    
    Док.Записать(РежимЗаписиДокумента.Проведение);
    
КонецПроцедуры
