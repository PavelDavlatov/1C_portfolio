1. Маршрут путешественника

Задача: 
Есть двумерный массив с билетами путешественника. Маршрут не может быть кольцевым. Расставить билеты по порядку.

Пример:
- Вход: [["Стамбул","Лондон"], ["Москва","Стамбул"], ["Лондон","Амстердам"]]
- Выход: [["Москва","Стамбул"], ["Стамбул","Лондон"], ["Лондон","Амстердам"]]

Решение:
&НаКлиенте
Процедура Старт(Команда)
    ПроверитьНаСервере();
КонецПроцедуры

&НаСервере
Процедура ПроверитьНаСервере()
    Билеты = Новый Массив;
    
    Билет = Новый Массив;
    Билет.Добавить("Москва");
    Билет.Добавить("Стамбул");
    Билеты.Добавить(Билет);
    
    Билет = Новый Массив;
    Билет.Добавить("Стамбул");
    Билет.Добавить("Лондон");
    Билеты.Добавить(Билет);
    
    Билет = Новый Массив;
    Билет.Добавить("Лондон");
    Билет.Добавить("Амстердам");
    Билеты.Добавить(Билет);
    
    Результат = РасставитьБилеты(Билеты);
    
    Для Каждого Билет Из Результат Цикл
        Сообщить(Билет[0] + " -> " + Билет[1]);
    КонецЦикла;
КонецПроцедуры

&НаСервереБезКонтекста
Функция РасставитьБилеты(Билеты)
    ГородаОтправления = Новый Соответствие;
    ГородаПрибытия = Новый Соответствие;
    Для Каждого Билет Из Билеты Цикл
        ГородаОтправления.Вставить(Билет[0], Билет);
        ГородаПрибытия.Вставить(Билет[1], Билет);
    КонецЦикла;
    Старт = "";
    Для Каждого Билет Из Билеты Цикл
        Если ГородаПрибытия[Билет[0]] = Неопределено Тогда
            Старт = Билет[0];
            Прервать;
        КонецЕсли;
    КонецЦикла;
    Результат = Новый Массив;
    ТекущийГород = Старт;
    Пока Истина Цикл
        Билет = ГородаОтправления[ТекущийГород];
        Если Билет = Неопределено Тогда
            Прервать;
        КонецЕсли;
        Результат.Добавить(Билет);
        ТекущийГород = Билет[1];
    КонецЦикла;
    Возврат Результат;
КонецФункции
