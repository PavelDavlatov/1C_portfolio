&НаСервере
Процедура СортироватьНаСервере()
    // Создание массива
    МассивЧисел = Новый Массив;
    МассивЧисел.Добавить(7);
    МассивЧисел.Добавить(1);
    МассивЧисел.Добавить(5);
    МассивЧисел.Добавить(3);
    МассивЧисел.Добавить(6);
    МассивЧисел.Добавить(4);
    
    // Сортировка методом "пузырька"
    Размер = МассивЧисел.Количество();
    Для Счетчик1 = 0 По Размер - 2 Цикл
        Для Счетчик2 = 0 По Размер - 2 - Счетчик1 Цикл
            Если МассивЧисел[Счетчик2] > МассивЧисел[Счетчик2 + 1] Тогда
                Буфер = МассивЧисел[Счетчик2];
                МассивЧисел[Счетчик2] = МассивЧисел[Счетчик2 + 1];
                МассивЧисел[Счетчик2 + 1] = Буфер;
            КонецЕсли;
        КонецЦикла;
    КонецЦикла;
    
    // Вывод результата в окно сообщений
    Сообщить("Отсортированный массив:");
    Для Каждого Элемент Из МассивЧисел Цикл
        Сообщить(Элемент);
    КонецЦикла;
КонецПроцедуры
