3. Максимальная разница в массиве

Задача:  
Найти максимальную разницу arr[j] - arr[i]`, где `i < j.

Пример:
- Вход:[7, 1, 5, 3, 6, 4] → Результат: 5 (6 - 1)

Решение:

&НаКлиенте
Процедура Запустить(Команда)
    ПроверитьЗадачу3();
КонецПроцедуры

&НаСервере
Процедура ПроверитьЗадачу3()
    Массив = Новый Массив;
    Массив.Добавить(7);
    Массив.Добавить(1);
    Массив.Добавить(5);
    Массив.Добавить(3);
    Массив.Добавить(6);
    Массив.Добавить(4);
    
    Результат = МаксимальнаяРазница(Массив);
    Сообщить("Максимальная разница: " + Результат);
КонецПроцедуры

&НаСервереБезКонтекста
Функция МаксимальнаяРазница(Массив)
    МинЭлемент = Массив[0];
    Результат = 0;
    
    Для Инд = 1 По Массив.ВГраница() Цикл
        Разница = Массив[Инд] - МинЭлемент;
        Если Разница > Результат Тогда
            Результат = Разница;
        КонецЕсли;
        Если Массив[Инд] < МинЭлемент Тогда
            МинЭлемент = Массив[Инд];
        КонецЕсли;
    КонецЦикла;
    
    Возврат Результат;
КонецФункции
