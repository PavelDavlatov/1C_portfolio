4. Дублирование нулей

Задача:
Дублировать каждое вхождение нуля, сдвигая остальные элементы вправо. Массив фиксированной длины.

Пример:
- Вход: [1, 0, 2, 3, 0, 4, 5, 0] → Выход: [1, 0, 0, 2, 3, 0, 0, 4]

Решение:

&НаКлиенте
Процедура Запустить(Команда)
    ПроверитьЗадачу4();
КонецПроцедуры

&НаСервере
Процедура ПроверитьЗадачу4()
    Массив = Новый Массив;
    Массив.Добавить(1);
    Массив.Добавить(0);
    Массив.Добавить(2);
    Массив.Добавить(3);
    Массив.Добавить(0);
    Массив.Добавить(4);
    Массив.Добавить(5);
    Массив.Добавить(0);
    
    ДублироватьНули(Массив);
    
    Строка = "";
    Для Каждого Эл Из Массив Цикл
        Строка = Строка + Эл + " ";
    КонецЦикла;
    Сообщить("Результат: " + Строка);
КонецПроцедуры

&НаСервереБезКонтекста
Процедура ДублироватьНули(Массив)
    ИсходнаяДлина = Массив.Количество();
    Инд = 0;
    
    Пока Инд < Массив.Количество() Цикл
        Если Массив[Инд] = 0 Тогда
            Массив.Вставить(Инд + 1, 0);
            Инд = Инд + 2;
        Иначе
            Инд = Инд + 1;
        КонецЕсли;
    КонецЦикла;
    
    Пока Массив.Количество() > ИсходнаяДлина Цикл
        Массив.Удалить(Массив.ВГраница());
    КонецЦикла;
КонецПроцедуры
