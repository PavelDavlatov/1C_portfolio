// ПРОЕКТ 2: Комплексная автоматизация ИТ-фирмы
// ОБЪЕКТ: Документ "ПоступлениеТоваровИУслуг", Модуль объекта
// НАЗНАЧЕНИЕ: Проведение поступления с разделением логики по типу номенклатуры:
//             товары приходят на склад, услуги списываются в расходы.
//             Формируются движения по регистрам накопления и бухгалтерии.
// =========================================================================

// Пересчет суммы документа перед записью (итог по табличной части)
Процедура ПередЗаписью(Отказ, РежимЗаписи, РежимПроведения)
    Сумма = ТоварыИУслуги.Итог("Сумма");
КонецПроцедуры

// Получение агрегированных данных из табличной части через запрос
// ЛЕВОЕ СОЕДИНЕНИЕ нужно для получения типа номенклатуры из справочника
// СГРУППИРОВАТЬ ПО устраняет дубли, если одна номенклатура встречается в нескольких строках
Функция ПолучитьТоварыИУслугиЗапросом(ДокументСсылка)   
    Запрос = Новый Запрос;
    Запрос.Текст =
    "ВЫБРАТЬ
    |   ПоступлениеТоваровИУслугТоварыИУслуги.Номенклатура КАК Номенклатура,
    |   СУММА(ПоступлениеТоваровИУслугТоварыИУслуги.Количество) КАК Количество,
    |   СУММА(ПоступлениеТоваровИУслугТоварыИУслуги.Сумма) КАК Сумма,
    |   СправочникНоменклатура.Тип КАК Тип
    |ИЗ
    |   Документ.ПоступлениеТоваровИУслуг.ТоварыИУслуги КАК ПоступлениеТоваровИУслугТоварыИУслуги
    |       ЛЕВОЕ СОЕДИНЕНИЕ Справочник.Номенклатура КАК СправочникНоменклатура
    |       ПО ПоступлениеТоваровИУслугТоварыИУслуги.Номенклатура = СправочникНоменклатура.Ссылка
    |ГДЕ
    |   ПоступлениеТоваровИУслугТоварыИУслуги.Ссылка = &Ссылка
    |
    |СГРУППИРОВАТЬ ПО
    |   ПоступлениеТоваровИУслугТоварыИУслуги.Номенклатура,
    |   СправочникНоменклатура.Тип";
    
    Запрос.УстановитьПараметр("Ссылка", ДокументСсылка);
    РезультатЗапроса = Запрос.Выполнить().Выгрузить();
    Возврат РезультатЗапроса;
КонецФункции 

// Основная процедура проведения: формирует движения по 4 регистрам
Процедура ОбработкаПроведения(Отказ, Режим)

    // Получаем значение перечисления для разделения товаров и услуг
    ТипТовар = ПредопределенноеЗначение("Перечисление.ТипыНоменклатуры.Товар");    
    
    // 1. Регистр накопления "ВзаиморасчетыСКонтрагентами" - фиксируем задолженность перед поставщиком
    Движения.ВзаиморасчетыСКонтрагентами.Записывать = Истина;
    Движение = Движения.ВзаиморасчетыСКонтрагентами.Добавить();
    Движение.ВидДвижения = ВидДвиженияНакопления.Расход;
    Движение.Период = Дата;
    Движение.Контрагент = Поставщик;
    Движение.Сумма = Сумма;

    // Подготавливаем регистры для записи в цикле
    Движения.Товары.Записывать = Истина;
    Движения.Расходы.Записывать = Истина;

    // 2. Регистр бухгалтерии "Управленческий" 
    Движения.Управленческий.Очистить();
    Движения.Управленческий.Записывать = Истина;    

    // Получаем агрегированные данные через запрос 
    ТЗТоварыИУслуги = ПолучитьТоварыИУслугиЗапросом(Ссылка);
    
    // Обходим каждую уникальную позицию номенклатуры
    Для Каждого ТекСтрокаТоварыИУслуги Из ТЗТоварыИУслуги Цикл                   

        Если ТекСтрокаТоварыИУслуги.Тип = ТипТовар Тогда
            // ТОВАРЫ: приход на склад (регистр "Товары")
            Движение = Движения.Товары.Добавить();
            Движение.ВидДвижения = ВидДвиженияНакопления.Приход;
            Движение.Период = Дата;
            Движение.Номенклатура = ТекСтрокаТоварыИУслуги.Номенклатура;
            Движение.Количество = ТекСтрокаТоварыИУслуги.Количество;
            Движение.Сумма = ТекСтрокаТоварыИУслуги.Сумма;

            // Проводка управленческого учета: Дт Товары - Кт Расчеты с поставщиками
            Движение = Движения.Управленческий.Добавить();
            Движение.Период = Дата;                                                         
            Движение.СчетДт = ПредопределенноеЗначение("ПланСчетов.Управленческий.Товары");
            Движение.СубконтоДт[ПланыВидовХарактеристик.ВидыСубконто.Номенклатура] = ТекСтрокаТоварыИУслуги.Номенклатура;
            Движение.СчетКт = ПредопределенноеЗначение("ПланСчетов.Управленческий.РасчетыСПоставщиками");
            Движение.СубконтоКт[ПланыВидовХарактеристик.ВидыСубконто.Контрагенты] = Поставщик;
            Движение.КоличествоДт = ТекСтрокаТоварыИУслуги.Количество;
            Движение.Сумма = ТекСтрокаТоварыИУслуги.Сумма;
            
        Иначе
            // УСЛУГИ/РАСХОДЫ: сразу списываем в регистр "Расходы" 
            Движение = Движения.Расходы.Добавить();
            Движение.Период = Дата;
            Движение.Номенклатура = ТекСтрокаТоварыИУслуги.Номенклатура;
            Движение.Сумма = ТекСтрокаТоварыИУслуги.Сумма;
        КонецЕсли;
        
    КонецЦикла;

КонецПроцедуры
