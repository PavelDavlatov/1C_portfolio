Процедура ОбработкаПроведения(Отказ, Режим)

	ТипТовар = ПредопределенноеЗначение("Перечисление.ТипыНоменклатуры.Товар");	
	
	Движения.ВзаиморасчетыСКонтрагентами.Записывать = Истина;
	
	Движение = Движения.ВзаиморасчетыСКонтрагентами.Добавить();
	Движение.ВидДвижения = ВидДвиженияНакопления.Расход;
	Движение.Период = Дата;
	Движение.Контрагент = Поставщик;
	Движение.Сумма = Сумма;

	Движения.Товары.Записывать = Истина;
	Движения.Расходы.Записывать = Истина;

	Движения.Управленческий.Очистить();
	Движения.Управленческий.Записывать = Истина;    

	ТЗТоварыИУслуги = ПолучитьТоварыИУслугиЗапросом(Ссылка);
	
	Для Каждого ТекСтрокаТоварыИУслуги Из ТЗТоварыИУслуги Цикл                   

		Если ТекСтрокаТоварыИУслуги.Тип = ТипТовар Тогда

			Движение = Движения.Товары.Добавить();
			Движение.ВидДвижения = ВидДвиженияНакопления.Приход;
			Движение.Период = Дата;
			Движение.Номенклатура = ТекСтрокаТоварыИУслуги.Номенклатура;
			Движение.Количество = ТекСтрокаТоварыИУслуги.Количество;
			Движение.Сумма = ТекСтрокаТоварыИУслуги.Сумма;

			Движение = Движения.Управленческий.Добавить();
			Движение.Период = Дата;                                                         
			Движение.СчетДт = ПредопределенноеЗначение("ПланСчетов.Управленческий.Товары");
			Движение.СубконтоДт[ПланыВидовХарактеристик.ВидыСубконто.Номенклатура] = ТекСтрокаТоварыИУслуги.Номенклатура;
			Движение.СчетКт = ПредопределенноеЗначение("ПланСчетов.Управленческий.РасчетыСПоставщиками");
			Движение.СубконтоКт[ПланыВидовХарактеристик.ВидыСубконто.Контрагенты] = Поставщик;
			Движение.КоличествоДт = ТекСтрокаТоварыИУслуги.Количество;
			Движение.Сумма = ТекСтрокаТоварыИУслуги.Сумма;
			
		Иначе

			Движение = Движения.Расходы.Добавить();
			Движение.Период = Дата;
			Движение.Номенклатура = ТекСтрокаТоварыИУслуги.Номенклатура;
			Движение.Сумма = ТекСтрокаТоварыИУслуги.Сумма;

		КонецЕсли;
		
	КонецЦикла;
