 Движения.Управленческий.Очистить();

 Движение.СчетДт = ПредопределенноеЗначение("ПланСчетов.Управленческий.ДенежныеСредства");
 
 Если ТипЗнч(Плательщик) = Тип("СправочникСсылка.Контрагенты") Тогда

  Движение.СубконтоКт[ПланыВидовХарактеристик.ВидыСубконто.Контрагенты] = Плательщик;
  
 ИначеЕсли ТипЗнч(Плательщик) = Тип("СправочникСсылка.Сотрудники") Тогда
  Движение.СчетКт = ПредопределенноеЗначение("ПланСчетов.Управленческий.РасчетыССотрудниками");
  Движение.СубконтоКт[ПланыВидовХарактеристик.ВидыСубконто.Сотрудники] = Плательщик;
  
 КонецЕсли;
 
 Движение.Сумма = Сумма;
