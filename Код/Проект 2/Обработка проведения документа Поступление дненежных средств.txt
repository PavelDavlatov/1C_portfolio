//             взаиморасчетам и регистру бухгалтерии (управленческий учет).
//             Поддерживает два типа плательщиков: Контрагенты и Сотрудники.
// =========================================================================

Процедура ОбработкаПроведения(Отказ, Режим)

    // 1. Регистр накопления "ВзаиморасчетыСКонтрагентами" - уменьшаем долг (расход)
    // Используем сокращенный синтаксис ДобавитьРасход() вместо явного указания ВидДвижения
    Движения.ВзаиморасчетыСКонтрагентами.Записывать = Истина;
    Движение = Движения.ВзаиморасчетыСКонтрагентами.ДобавитьРасход();
    Движение.Период = Дата;
    Движение.Контрагент = Плательщик;
    Движение.Сумма = Сумма;

    // 2. Регистр бухгалтерии "Управленческий" - формируем проводку Дт ДС - Кт Расчеты
    Движения.Управленческий.Очистить();
    Движения.Управленческий.Записывать = Истина;
    
    Движение = Движения.Управленческий.Добавить();
    Движение.Период = Дата;
    // Дебет всегда "Денежные средства" - деньги поступают на счет
    Движение.СчетДт = ПредопределенноеЗначение("ПланСчетов.Управленческий.ДенежныеСредства");
    
    // Кредит зависит от типа плательщика: разные счета учета для разных контрагентов
    Если ТипЗнч(Плательщик) = Тип("СправочникСсылка.Контрагенты") Тогда
        // Поступление от покупателя -> закрываем задолженность по расчетам с покупателями
        Движение.СчетКт = ПредопределенноеЗначение("ПланСчетов.Управленческий.РасчетыСПокупателями");
        Движение.СубконтоКт[ПланыВидовХарактеристик.ВидыСубконто.Контрагенты] = Плательщик;
        
    ИначеЕсли ТипЗнч(Плательщик) = Тип("СправочникСсылка.Сотрудники") Тогда
        // Поступление от сотрудника (например, возврат подотчетных средств)
        Движение.СчетКт = ПредопределенноеЗначение("ПланСчетов.Управленческий.РасчетыССотрудниками");
        Движение.СубконтоКт[ПланыВидовХарактеристик.ВидыСубконто.Сотрудники] = Плательщик;
        
    КонецЕсли;
    
    Движение.Сумма = Сумма;
    
КонецПроцедуры
